# The Thermal Station's screen (ThermalGauge) in the ranges ticket #64 named.
# From the repo root: tools/screenshot/shot.sh --sheet tools/screenshot/recipes/thermal.txt
# Full power in the sand with all heat sent outward warms the hull through every range in 90 s.
# Debug commands: none.

# At a fifth of full power: the hull well under the amber range, the bar white, no lamp lit.
cool: --look Thermal --time 20 --do "0 set_power 0.2"

# The hull inside the amber range: the bar and the number amber, HULL HOT lit.
caution range: --look Thermal --time 58 --do "0 set_power 1"

# The hull past what it bears: the bar and the number red, integrity falling.
warning range: --look Thermal --time 90 --do "0 set_power 1"

# Most of the heat sent into the Column from 60 s: the split bar divided at the cyan mark, the trend FALLING.
heat sent to the Column: --look Thermal --time 75 --do "0 set_power 1" --do "60 set_heat_split 0.7"
